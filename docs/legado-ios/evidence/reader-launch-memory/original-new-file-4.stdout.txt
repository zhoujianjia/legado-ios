+def memory_row(fixture, iteration, duration=0.029, ticks=(0.0105, 0.0205)):
+    base = 100.0 + iteration
+    events = [{"name": name, "uptimeSeconds": base + duration * i / (len(guard.STAGES) - 1)}
+              for i, name in enumerate(guard.STAGES)]
+    points = [{"kind": "boundary", "phase": event["name"], "beforeUptimeSeconds": event["uptimeSeconds"] + 0.00001,
+               "afterUptimeSeconds": event["uptimeSeconds"] + 0.000011} for event in events]
+    for tick in ticks:
+        phase = [event["name"] for event in events if event["uptimeSeconds"] <= base + tick][-1]
+        points.append({"kind": "periodic", "phase": phase,
+                       "beforeUptimeSeconds": base + tick, "afterUptimeSeconds": base + tick + 0.000001})
+    points.sort(key=lambda point: point["beforeUptimeSeconds"])
+    for index, point in enumerate(points):
+        point.update(footprintBytes=10_000_000 + index * 100, residentBytes=12_000_000 + index * 200,
+                     kernelReturn=0, sufficientReturnCount=True, onMain=False)
+    periodic = [point for point in points if point["kind"] == "periodic"]
+    timeline = [points[0]] + periodic + [points[-1]]
+    measurement = {"processID": 7000, "beginUptimeSeconds": points[0]["beforeUptimeSeconds"],
+                   "endUptimeSeconds": points[-1]["afterUptimeSeconds"], "complete": True, "errors": [],
+                   "periodicSamples": len(periodic), "maximumObservedGapMilliseconds": max(
+                       (b["beforeUptimeSeconds"] - a["beforeUptimeSeconds"]) * 1000 for a, b in zip(timeline, timeline[1:])),
+                   "baselineFootprintBytes": points[0]["footprintBytes"], "baselineResidentBytes": points[0]["residentBytes"],
+                   "sampledPeakFootprintBytes": max(point["footprintBytes"] for point in points),
+                   "sampledPeakResidentBytes": max(point["residentBytes"] for point in points), "points": points, "events": events}

+    anchor = ("讀書使人充實，思考使人深邃，交談使人清醒。" * 2)[:32]

+def memory_rows():

+def phases(rows):
+    return [{"runID": row["runID"], "fixture": row["fixture"], "iteration": row["iteration"], "event": event}
+            for row in rows for event in row["measurement"]["events"]]

+class LaunchMemoryShapeGuards(unittest.TestCase):
+    def test_owned_complete_shapes_are_distinct_from_execution_attestation(self):
+        self.assertEqual(len(guard.validate_launch(launch_rows())), 5)
+        rows = memory_rows()
+        self.assertEqual(len(guard.validate_memory(rows, phases(rows))), 10)
+
+    def test_existing_100k_fixture_protocol_is_reused_and_1m_is_separate(self):
+        spec = importlib.util.spec_from_file_location("existing_stage_fixture_protocol", HERE / "reader-stage-baseline.py")
+        old = importlib.util.module_from_spec(spec); spec.loader.exec_module(old)
+        self.assertEqual(guard.fixture_protocol()["plain-100k"], old.fixture_protocol()["plain-100k"])
+        self.assertEqual(set(guard.fixture_protocol()), {"plain-100k", "plain-1m"})
+        self.assertEqual(guard.fixture_protocol()["plain-1m"]["fixtureMinimumUTF16"], 1_048_576)
+
+    def test_launch_requires_exact_five_distinct_samples_and_process_identities(self):
+        for mutation in (lambda rows: rows.pop(), lambda rows: rows.append(copy.deepcopy(rows[0])),
+                         lambda rows: rows[1].update(iteration=0),
+                         lambda rows: rows[1]["observed"].update(processObservationID=rows[0]["observed"]["processObservationID"]),
+                         lambda rows: rows[0]["observed"].update(processObservationID=rows[0]["expected"]["processObservationID"])):
+            rows = launch_rows(); mutation(rows)
+            with self.subTest(mutation=mutation), self.assertRaises(ValueError): guard.validate_launch(rows)
+        with self.assertRaises(ValueError): guard.validate_launch([])
+        rows = launch_rows()
+        rows[1]["observed"]["processID"] = rows[0]["observed"]["processID"]
+        self.assertEqual(len(guard.validate_launch(rows)), 5, "PID reuse alone is not process reuse")
+
+    def test_launch_cannot_claim_import_free_device_cold_or_pixels_without_flags(self):
+        for key in ("fixturePreparationInsideInterval", "measuredLaunchImports", "systemCachesPurged", "deviceCold", "gpuPresentationMeasured", "improvementEstablished"):
+            rows = launch_rows(); rows[0][key] = True
+            with self.subTest(key=key), self.assertRaises(ValueError): guard.validate_launch(rows)
+        rows = launch_rows(); rows[0]["originalImportInputRemoved"] = 1
+        with self.assertRaises(ValueError): guard.validate_launch(rows)
+
+    def test_launch_requires_the_actual_utf16_text_range_and_unique_marker(self):
+        for key, value in (("pageStartUTF16", 1), ("pageLengthUTF16", 1), ("pageText", "wrong text"), ("viewBindingRevision", True)):
+            rows = launch_rows(); rows[0]["observed"][key] = value
+            with self.subTest(key=key), self.assertRaises(ValueError): guard.validate_launch(rows)
+        self.assertEqual(guard.utf16("a😀b"), 4)
+        self.assertNotEqual("e\u0301", "é")
+
+    def test_launch_missing_reversed_or_nonfinite_timestamps_fail(self):
+        for value in (True, float("nan"), float("inf"), -1):
+            rows = launch_rows(); rows[0]["timestamps"]["launchRequested"] = value
+            with self.subTest(value=value), self.assertRaises(ValueError): guard.validate_launch(rows)
+        rows = launch_rows(); rows[0]["timestamps"].pop("pageObserved")
+        with self.assertRaises(ValueError): guard.validate_launch(rows)
+
+    def test_memory_requires_exact_two_by_five_matrix(self):
+        for mutation in (lambda rows: rows.pop(), lambda rows: rows.append(copy.deepcopy(rows[0])),
+                         lambda rows: rows[0].update(fixture="plain-10k"), lambda rows: rows[0].update(iteration=True)):
+            rows = memory_rows(); mutation(rows)
+            with self.subTest(mutation=mutation), self.assertRaises(ValueError): guard.validate_memory(rows, phases(rows))
+
+    def test_raw_phase_markers_must_be_complete_unique_and_ordered(self):
+        rows = memory_rows()
+        for mutation in (lambda records: records.pop(), lambda records: records.append(copy.deepcopy(records[0])),
+                         lambda records: records.reverse()):
+            raw = phases(rows); mutation(raw)
+            with self.subTest(mutation=mutation), self.assertRaises(ValueError): guard.validate_memory(rows, raw)
+
+    def test_failed_on_main_or_truncated_kernel_queries_cannot_be_peaks(self):
+        for key, value in (("kernelReturn", 5), ("kernelReturn", False), ("sufficientReturnCount", False),
+                           ("onMain", True), ("footprintBytes", -1), ("residentBytes", None)):
+            rows = memory_rows(); rows[0]["measurement"]["points"][0][key] = value
+            with self.subTest(key=key), self.assertRaises(ValueError): guard.validate_memory(rows, phases(rows))
+
+    def test_missing_periodic_samples_or_false_counts_fail(self):
+        rows = memory_rows()
+        row = rows[0]["measurement"]
+        row["points"] = [point for point in row["points"] if point["kind"] != "periodic"]
+        row["periodicSamples"] = 0
+        with self.assertRaises(ValueError): guard.validate_memory(rows, phases(rows))
+        rows = memory_rows(); rows[0]["measurement"]["periodicSamples"] = True
+        with self.assertRaises(ValueError): guard.validate_memory(rows, phases(rows))
+        rows = memory_rows()
+        next(point for point in rows[0]["measurement"]["points"] if point["kind"] == "periodic")["phase"] = "baseline"
+        with self.assertRaises(ValueError): guard.validate_memory(rows, phases(rows))
+
+    def test_boundary_samples_cannot_hide_a_missed_periodic_gap(self):

+        rows[0]["measurement"]["maximumObservedGapMilliseconds"] = 4

+    def test_baseline_and_peak_must_match_original_raw_points(self):

+    def test_cli_retains_a_failure_record_without_attesting_runtime_or_overwriting(self):
+        with tempfile.TemporaryDirectory(prefix="owned-launch-memory-shapes-") as temporary:
+            directory = Path(temporary)
+            launch = directory / "owned-synthetic-launch.log"; memory = directory / "owned-synthetic-memory.log"
+            launch.write_text("\n".join(guard.LAUNCH + json.dumps(row, ensure_ascii=False) for row in launch_rows()) + "\n", encoding="utf-8")
+            rows = memory_rows()
+            memory.write_text("\n".join(guard.PHASE + json.dumps(row) for row in phases(rows)) + "\n" +
+                              "\n".join(guard.MEMORY + json.dumps(row, ensure_ascii=False) for row in rows) + "\n", encoding="utf-8")
+            output = directory / "owned-synthetic-shape-only.json"
+            command = [sys.executable, "-B", str(HERE / "reader-launch-memory.py"), "--launch-log", str(launch), "--memory-log", str(memory), "--output", str(output)]
+            result = subprocess.run(command, capture_output=True)
+            self.assertEqual(result.returncode, 0, result.stderr)
+            report = json.loads(output.read_bytes())
+            self.assertTrue(report["collectionShapeValidated"])
+            self.assertFalse(report["sdkCompilationAttested"])
+            self.assertFalse(report["iOSExecutionAttested"])
+            before = output.read_bytes()
+            self.assertNotEqual(subprocess.run(command, capture_output=True).returncode, 0)
+            self.assertEqual(output.read_bytes(), before)
+            launch.write_bytes(b"")
+            command[-1] = str(directory / "failed-shape-only.json")
+            self.assertEqual(subprocess.run(command, capture_output=True).returncode, 1)
+            failed = json.loads(Path(command[-1]).read_bytes())
+            self.assertFalse(failed["collectionShapeValidated"])
+            self.assertFalse(failed["iOSExecutionAttested"])
